cmake_minimum_required(VERSION 3.27...4.1)
project(m2fs_navigation_probe VERSION 0.1 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED Celeritas_DIR)
  message(FATAL_ERROR "Pass the verified h5 install/lib64/cmake/Celeritas explicitly")
endif()
find_package(Celeritas 0.6.3 EXACT REQUIRED)
find_package(Geant4 11.3.2 EXACT REQUIRED)
if(NOT CELERITAS_USE_CUDA OR NOT CELERITAS_USE_covfie OR NOT CELERITAS_USE_VecGeom)
  message(FATAL_ERROR "Requires the existing CUDA/covfie/VecGeom h5 configuration")
endif()
set(M2FS_CARTMAP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Pinned h4 parser directory")
add_executable(m2fs-navigation-probe probe.cc NavKernels.cu "${M2FS_CARTMAP_DIR}/CartMapIO.cc")
target_include_directories(m2fs-navigation-probe PRIVATE "${M2FS_CARTMAP_DIR}" ${Geant4_INCLUDE_DIRS})
celeritas_target_link_libraries(m2fs-navigation-probe PRIVATE
  Celeritas::celeritas Celeritas::geocel Celeritas::Extcovfie VecGeom::vecgeom CUDA::cudart
  nlohmann_json::nlohmann_json ${Geant4_LIBRARIES})
get_target_property(_rdc m2fs-navigation-probe CUDA_SEPARABLE_COMPILATION)
if(NOT _rdc)
  message(FATAL_ERROR "Celeritas RDC wrapper did not enable separable CUDA compilation")
endif()
